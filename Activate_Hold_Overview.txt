BEGIN;
CREATE OR REPLACE FUNCTION public.rr_reward_overview(p_org uuid,p_program uuid,p_period uuid DEFAULT NULL) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $$
DECLARE result jsonb;BEGIN
 PERFORM public.rr_motivation_require_access(p_org);
 IF NOT EXISTS(SELECT 1 FROM public.rr_motivation_programs WHERE id=p_program AND organization_id=p_org) THEN RAISE EXCEPTION 'Program not found' USING ERRCODE='42501';END IF;
 IF p_period IS NOT NULL AND NOT EXISTS(SELECT 1 FROM public.rr_reward_periods WHERE id=p_period AND program_id=p_program AND organization_id=p_org) THEN RAISE EXCEPTION 'Week not found' USING ERRCODE='42501';END IF;
 WITH awarded AS(SELECT student_id,sum(points) approved FROM public.rr_motivation_points WHERE organization_id=p_org AND program_id=p_program GROUP BY student_id),
 redeemed AS(SELECT d.student_id,sum(round(d.base_points*d.pct/100,4)) approved,sum(d.base_points-round(d.base_points*d.pct/100,4)) held FROM public.rr_reward_redemptions d JOIN public.rr_motivation_cards c ON c.id=d.card_id WHERE d.organization_id=p_org AND c.program_id=p_program AND (p_period IS NULL OR d.period_id=p_period) GROUP BY d.student_id)
 SELECT coalesce(jsonb_agg(jsonb_build_object('id',s.id,'name',s.name,'class_id',a.class_id,'approved',CASE WHEN p_period IS NULL THEN coalesce(w.approved,0) ELSE coalesce(r.approved,0) END,'held',coalesce(r.held,0),'state',CASE WHEN p_period IS NULL THEN NULL WHEN permissions.minimum>=100 THEN 'approved' WHEN coalesce(permissions.maximum,0)<=0 THEN 'hold' ELSE 'partial' END)),'[]') INTO result
 FROM public.students s LEFT JOIN public.student_class_assignments a ON a.student_id=s.id AND a.organization_id=p_org AND a.is_current LEFT JOIN awarded w ON w.student_id=s.id LEFT JOIN redeemed r ON r.student_id=s.id
 LEFT JOIN LATERAL (SELECT min(public.rr_reward_percent(c,s.id)) minimum,max(public.rr_reward_percent(c,s.id)) maximum FROM public.rr_motivation_cards c WHERE c.organization_id=p_org AND c.program_id=p_program AND c.period_id=p_period AND c.class_id=a.class_id) permissions ON p_period IS NOT NULL
 WHERE s.organization_id=p_org AND NOT s.is_archived;
 RETURN jsonb_build_object('students',result,'approved',coalesce((SELECT sum((value->>'approved')::numeric) FROM jsonb_array_elements(result)),0),'held',coalesce((SELECT sum((value->>'held')::numeric) FROM jsonb_array_elements(result)),0));
END $$;
REVOKE ALL ON FUNCTION public.rr_reward_overview(uuid,uuid,uuid) FROM PUBLIC,anon,authenticated;
GRANT EXECUTE ON FUNCTION public.rr_reward_overview(uuid,uuid,uuid) TO authenticated;
NOTIFY pgrst,'reload schema';
COMMIT;
SELECT 'MOTIVATION_HOLD_OVERVIEW_READY' AS result;
